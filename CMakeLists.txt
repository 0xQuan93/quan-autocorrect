cmake_minimum_required(VERSION 3.16)
project(quan-autocorrect LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Fcitx5Core 5.1.22 REQUIRED)
find_package(Fcitx5ModuleSpell REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_library(quan-autocorrect MODULE autocorrect.cpp)
target_link_libraries(quan-autocorrect PRIVATE Fcitx5::Core Fcitx5::Module::Spell)
target_compile_options(quan-autocorrect PRIVATE -Wall -Wextra)
add_executable(test-autocorrect test.cpp)
target_link_libraries(test-autocorrect PRIVATE Fcitx5::Core Fcitx5::Module::Spell)
enable_testing()
add_test(NAME typing COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test.py $<TARGET_FILE:test-autocorrect>)
